    <script>
        // --- CONFIGURACIÓN ---
        // 1. Obtén tu clave en https://aistudio.google.com/
        // 2. Pégala aquí abajo entre las comillas.
        const apiKey = ""; 

        const videos = [
            { id: 1, title: "¿Cómo ingreso a Canvas?", embedUrl: "https://www.youtube.com/embed/lf777UQVUBg", category: "Primeros Pasos", subtopics: [{l: "Configuración inicial", t: 15}, {l: "Acceso", t: 45}] },
            { id: 2, title: "Página de Inicio", embedUrl: "https://www.youtube.com/embed/3nlkadip6kg", category: "Primeros Pasos", subtopics: [{l: "Menú lateral", t: 30}] },
            { id: 8, title: "Entrega de Tareas", embedUrl: "https://www.youtube.com/embed/kt3gOU4q7ko", category: "Entregas y Notas", subtopics: [{l: "Carga de archivos", t: 75}] },
            { id: 9, title: "Mis Calificaciones", embedUrl: "https://www.youtube.com/embed/RjSSbQL-7s4", category: "Entregas y Notas", subtopics: [{l: "Feedback", t: 80}] }
        ];

        async function callGemini(prompt, system) {
            if (!apiKey) {
                return "ERROR_NO_KEY";
            }
            const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
            try {
                const res = await fetch(url, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: prompt }] }],
                        systemInstruction: { parts: [{ text: system }] }
                    })
                });
                const data = await res.json();
                return data.candidates[0].content.parts[0].text;
            } catch (e) {
                return "ERROR_API";
            }
        }

        async function sendChatMessage() {
            const input = document.getElementById('aiChatInput');
            const messages = document.getElementById('chatMessages');
            const query = input.value.trim();
            if(!query) return;

            messages.innerHTML += `<div class="bg-[#004a45] text-white p-4 rounded-2xl text-sm ml-auto max-w-[85%]">${query}</div>`;
            input.value = "";
            
            const loadingId = 'loading-' + Date.now();
            messages.innerHTML += `<div id="${loadingId}" class="bg-white p-4 rounded-2xl text-sm border border-slate-200">Pensando...</div>`;
            messages.scrollTop = messages.scrollHeight;

            const system = `Eres un asistente de Canvas Tecmilenio. Base de datos: ${JSON.stringify(videos)}. Responde con el video exacto e incluye este enlace: <span class="ai-link" onclick="jumpToVideo(ID, SEGUNDOS)">Ver video</span>.`;
            const resp = await callGemini(query, system);

            const bubble = document.getElementById(loadingId);
            if(resp === "ERROR_NO_KEY") {
                bubble.innerHTML = `<div class="error-bubble"><strong>Error:</strong> No has configurado tu API Key. Edita el archivo index.html y pega tu clave de Google AI Studio en la línea 133.</div>`;
            } else {
                bubble.innerHTML = resp.replace(/\n/g, '<br>');
            }
            messages.scrollTop = messages.scrollHeight;
        }

        window.jumpToVideo = (id, t) => {
            const v = videos.find(x => x.id == id);
            if(v) openVideo(v, t);
        };

        function openVideo(v, t=0) {
            document.getElementById('modalTitle').innerText = v.title;
            document.getElementById('videoPlayer').src = `${v.embedUrl}?autoplay=1&start=${t}`;
            document.getElementById('videoModal').classList.add('active');
            document.body.style.overflow = 'hidden';
        }

        function closeVideo() {
            document.getElementById('videoPlayer').src = "";
            document.getElementById('videoModal').classList.remove('active');
            document.body.style.overflow = 'auto';
        }

        function toggleAIChat() { document.getElementById('aiChatPanel').classList.toggle('active'); }

        function renderVideos() {
            const grid = document.getElementById('videoGrid');
            grid.innerHTML = videos.map(v => `
                <div class="video-card bg-white p-6 shadow-sm border border-slate-100 flex flex-col">
                    <span class="text-[10px] font-bold text-[#00a859] mb-2 uppercase">${v.category}</span>
                    <h3 class="font-bold text-[#004a45] mb-4">${v.title}</h3>
                    <button onclick='openVideo(${JSON.stringify(v)})' class="bg-[#004a45] text-white py-3 rounded-xl font-bold text-sm hover:bg-[#00a859] transition-all mt-auto">Ver Tutorial</button>
                </div>
            `).join('');
        }

        document.addEventListener('DOMContentLoaded', renderVideos);
    </script>
